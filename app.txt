from flask import Flask, render_template, request, jsonify
from flask_socketio import SocketIO, emit
import anthropic
import os
import json
from datetime import datetime
from dotenv import load_dotenv

load_dotenv()  # 👈 This loads .env file

app = Flask(__name__)
app.config['SECRET_KEY'] = 'student-chat-secret-2024'

client = anthropic.Anthropic(
    api_key=os.getenv("ANTHROPIC_API_KEY")
)

app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024
socketio = SocketIO(app, cors_allowed_origins="*", async_mode='threading')

active_users = {}
chat_history = []

SYSTEM_PROMPT = """You are StudyBot 🎓 — a friendly AI study assistant for students.
- Explain concepts clearly with step-by-step breakdowns
- Use examples and analogies
- Support Tamil and English (code-switch if student writes in Tamil)
- Be warm, encouraging, and educational
- Format answers with numbered steps when needed
- Start response with a relevant emoji"""

# ─── Routes ─────────────────────────────────────

@app.route('/')
def index():
    return render_template('index.html')


@app.route('/upload-pdf', methods=['POST'])
def upload_pdf():
    try:
        data = request.json
        pdf_text = data.get('text', '').strip()
        filename = data.get('filename', 'document.pdf')

        if not pdf_text:
            return jsonify({'error': 'No text found in PDF'}), 400

        if len(pdf_text) > 12000:
            pdf_text = pdf_text[:12000] + "\n\n[... truncated ...]"

        response = client.messages.create(
            model="claude-opus-4-6",
            max_tokens=1500,
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content":
                f'PDF: "{filename}"\n\nContent:\n{pdf_text}\n\n'
                f'Please: 1) Brief summary (3-4 sentences) 2) Top 5 key points 3) 3 exam questions from this'}]
        )

        summary = response.content[0].text
        bot_msg = {
            'type': 'bot', 'username': 'StudyBot 🎓',
            'message': f'📄 PDF Uploaded: {filename}\n\n{summary}',
            'timestamp': datetime.now().strftime('%H:%M'), 'sid': 'bot'
        }
        chat_history.append(bot_msg)
        socketio.emit('new_message', bot_msg)
        return jsonify({'success': True, 'summary': summary})

    except Exception as e:
        return jsonify({'error': str(e)}), 500


@app.route('/generate-quiz', methods=['POST'])
def generate_quiz():
    try:
        data = request.json
        topic = data.get('topic', '').strip()
        subject = data.get('subject', 'General')
        count = min(int(data.get('count', 5)), 10)

        if not topic:
            return jsonify({'error': 'Topic required'}), 400

        response = client.messages.create(
            model="claude-opus-4-6",
            max_tokens=2500,
            system="""Generate MCQ quiz as pure JSON only. No markdown, no extra text. Format:
{"questions":[{"q":"Question?","options":["A) opt1","B) opt2","C) opt3","D) opt4"],"answer":"A","explanation":"Why A is correct"}]}""",
            messages=[{"role": "user", "content":
                f'Generate {count} MCQ questions on "{topic}" for {subject} students. Medium difficulty.'}]
        )

        raw = response.content[0].text.strip()
        if '```' in raw:
            raw = raw.split('```')[1].lstrip('json').strip()
            if '```' in raw:
                raw = raw.split('```')[0].strip()

        quiz_data = json.loads(raw)
        return jsonify({'success': True, 'quiz': quiz_data, 'topic': topic, 'subject': subject})

    except Exception as e:
        return jsonify({'error': str(e)}), 500


# ─── Socket Events ──────────────────────────────

@socketio.on('connect')
def on_connect():
    print(f"Connected: {request.sid}")

@socketio.on('disconnect')
def on_disconnect():
    if request.sid in active_users:
        username = active_users.pop(request.sid)
        emit('user_left', {'username': username, 'active_count': len(active_users)}, broadcast=True)

@socketio.on('join')
def on_join(data):
    username = data.get('username', 'Student')
    active_users[request.sid] = username
    emit('chat_history', {'messages': chat_history[-50:]})
    emit('user_joined', {'username': username, 'active_count': len(active_users)}, broadcast=True)
    emit('active_users', {'users': list(active_users.values())})

@socketio.on('send_message')
def on_message(data):
    username = active_users.get(request.sid, 'Unknown')
    message = data.get('message', '').strip()
    if not message:
        return

    timestamp = datetime.now().strftime('%H:%M')
    msg_obj = {'type': 'user', 'username': username, 'message': message,
                'timestamp': timestamp, 'sid': request.sid}
    chat_history.append(msg_obj)
    emit('new_message', msg_obj, broadcast=True)

    if message.lower().startswith('@bot') or message.lower().startswith('bot,') or message.endswith('?'):
        emit('bot_typing', {'typing': True}, broadcast=True)
        try:
            query = message.replace('@bot', '').replace('bot,', '').strip() or message
            response = client.messages.create(
                model="claude-opus-4-6", max_tokens=1024,
                system=SYSTEM_PROMPT,
                messages=[{"role": "user", "content": query}]
            )
            bot_reply = response.content[0].text
            bot_msg = {'type': 'bot', 'username': 'StudyBot 🎓', 'message': bot_reply,
                       'timestamp': datetime.now().strftime('%H:%M'), 'sid': 'bot'}
            chat_history.append(bot_msg)
            emit('bot_typing', {'typing': False}, broadcast=True)
            emit('new_message', bot_msg, broadcast=True)
        except Exception as e:
            emit('bot_typing', {'typing': False}, broadcast=True)
            emit('new_message', {'type': 'bot', 'username': 'StudyBot 🎓',
                 'message': f'⚠️ Error: {str(e)}',
                 'timestamp': datetime.now().strftime('%H:%M'), 'sid': 'bot'}, broadcast=True)

@socketio.on('ask_bot')
def on_ask_bot(data):
    username = active_users.get(request.sid, 'Student')
    question = data.get('question', '').strip()
    subject = data.get('subject', 'General')
    if not question:
        return

    user_msg = {'type': 'user', 'username': username,
                'message': f'[{subject}] {question}',
                'timestamp': datetime.now().strftime('%H:%M'), 'sid': request.sid}
    chat_history.append(user_msg)
    emit('new_message', user_msg, broadcast=True)
    emit('bot_typing', {'typing': True}, broadcast=True)

    try:
        response = client.messages.create(
            model="claude-opus-4-6", max_tokens=1500,
            system=SYSTEM_PROMPT,
            messages=[{"role": "user", "content": f"Subject: {subject}\nQuestion: {question}"}]
        )
        bot_reply = response.content[0].text
        bot_msg = {'type': 'bot', 'username': 'StudyBot 🎓', 'message': bot_reply,
                   'timestamp': datetime.now().strftime('%H:%M'), 'sid': 'bot'}
        chat_history.append(bot_msg)
        emit('bot_typing', {'typing': False}, broadcast=True)
        emit('new_message', bot_msg, broadcast=True)
    except Exception as e:
        emit('bot_typing', {'typing': False}, broadcast=True)
        emit('error', {'message': str(e)})

function pdfDragOver(e){ e.preventDefault(); document.getElementById('pdfDrop').classList.add('drag-over'); }
function pdfDragLeave(){ document.getElementById('pdfDrop').classList.remove('drag-over'); }
function pdfDrop_ev(e){
  e.preventDefault();
  pdfDragLeave();
  const file = e.dataTransfer.files[0];
  if(file && file.type==='application/pdf') handlePdfFile(file);
}

async function handlePdfFile(file){
  if(!file) return;
  const status = document.getElementById('pdfStatus');
  status.className='pdf-status loading';
  status.textContent='📄 Reading PDF...';

  try{
    const arrayBuffer = await file.arrayBuffer();
    const pdf = await pdfjsLib.getDocument({data: arrayBuffer}).promise;
    let fullText = '';

    for(let i=1; i<=Math.min(pdf.numPages, 30); i++){
      const page = await pdf.getPage(i);
      const content = await page.getTextContent();
      const pageText = content.items.map(item => item.str).join(' ');
      fullText += pageText + '\n\n';
    }

    if(!fullText.trim()){
      status.className='pdf-status err';
      status.textContent='❌ Could not extract text (scanned PDF?)';
      return;
    }

    status.textContent='🤖 StudyBot is analyzing...';

    const response = await fetch('/upload-pdf', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body: JSON.stringify({text: fullText, filename: file.name, username: myUsername})
    });
    const data = await response.json();

    if(data.success){
      status.className='pdf-status done';
      status.textContent=`✅ "${file.name}" analyzed! Check the chat.`;
    } else {
      status.className='pdf-status err';
      status.textContent='❌ Error: '+data.error;
    }
  } catch(err){
    status.className='pdf-status err';
    status.textContent='❌ Failed to read PDF: '+err.message;
  }
}

let quizData = null;
let quizAnswers = {};
let quizSubmitted = false;

async function generateQuiz(){
  const topic = document.getElementById('quizTopic').value.trim();
  const subject = document.getElementById('quizSubject').value.trim() || 'General';
  const count = document.getElementById('quizCount').value;

  if(!topic){ document.getElementById('quizTopic').focus(); return; }

  const btn = document.querySelector('.action-btn.gold');
  btn.textContent='⏳ Generating...';
  btn.disabled=true;

  try{
    const response = await fetch('/generate-quiz',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body: JSON.stringify({topic, subject, count})
    });
    const data = await response.json();

    if(data.success){
      quizData = data.quiz;
      quizAnswers = {};
      quizSubmitted = false;
      showQuiz(data.topic, data.subject);
    } else {
      alert('Error: '+data.error);
    }
  } catch(e){
    alert('Network error: '+e.message);
  } finally{
    btn.textContent='🧠 Generate Quiz';
    btn.disabled=false;
  }
}

  const body = document.getElementById('quizBody');

function selectAnswer(qi, letter){
  if(quizSubmitted) return;
  quizAnswers[qi] = letter;
  const opts = document.querySelectorAll(`#opts-${qi} .quiz-opt`);
  opts.forEach(o => o.classList.remove('selected'));
  document.getElementById(`opt-${qi}-${letter}`).classList.add('selected');
}

function submitQuiz(){
  if(quizSubmitted) return;
  quizSubmitted = true;

  let score = 0;
  quizData.questions.forEach((q, qi)=>{
    const chosen = quizAnswers[qi];
    const correct = q.answer;
    if(chosen === correct) score++;

    // Mark options
    const opts = document.querySelectorAll(`#opts-${qi} .quiz-opt`);
    opts.forEach(o => {
      const letter = o.id.split('-')[2];
      if(letter===correct) o.classList.add('correct');
      else if(chosen && letter===chosen && chosen!==correct) o.classList.add('wrong');
    });

    // Show explanation
    document.getElementById(`exp-${qi}`).classList.add('show');
  });

  // Show score
  const total = quizData.questions.length;
  const pct = Math.round(score/total*100);
  const emoji = pct>=80?'🎉':pct>=60?'👍':pct>=40?'📚':'💪';

  const scoreDiv = document.createElement('div');
  scoreDiv.className='quiz-score';
  scoreDiv.innerHTML=`
    <div class="quiz-score-num">${score}/${total}</div>
    <div class="quiz-score-label">${pct}% — ${emoji} ${pct>=80?'Excellent!':pct>=60?'Good job!':pct>=40?'Keep studying!':'Keep practicing!'}</div>
    <button class="action-btn" style="max-width:180px;margin-top:16px;" onclick="closeQuiz()">Close</button>
  `;
  document.getElementById('quizBody').appendChild(scoreDiv);

  // Post score to chat
  socket.emit('send_message', {
    message: `🧠 Quiz Result: "${document.getElementById('quizTitle').textContent.replace('🧠 ','')}" — Scored ${score}/${total} (${pct}%) ${emoji}`
  });
}

// ─── Pomodoro Timer ─────────────────────────
const TIMER_MODES = {focus:{mins:25,label:'FOCUS'},short:{mins:5,label:'SHORT BREAK'},long:{mins:15,label:'LONG BREAK'}};
let currentMode = 'focus';
let timerSeconds = 25*60;
let timerMax = 25*60;
let timerRunning = false;
let timerInterval = null;
let sessionCount = 0;

function setTimerMode(mode){
  if(timerRunning) return;
  currentMode = mode;
  document.querySelectorAll('.mode-btn').forEach(b => b.classList.remove('active'));
  document.getElementById('mode-'+mode).classList.add('active');
  timerSeconds = TIMER_MODES[mode].mins * 60;
  timerMax = timerSeconds;
  document.getElementById('timerLabel').textContent = TIMER_MODES[mode].label;
  updateTimerDisplay();
}

function toggleTimer(){
  timerRunning = !timerRunning;
  const btn = document.getElementById('timerStartBtn');
  if(timerRunning){
    btn.textContent='⏸ Pause';
    timerInterval = setInterval(timerTick, 1000);
  } else {
    btn.textContent='▶ Resume';
    clearInterval(timerInterval);
  }
}

function timerTick(){
  if(timerSeconds <= 0){
    clearInterval(timerInterval);
    timerRunning = false;
    document.getElementById('timerStartBtn').textContent='▶ Start';
    if(currentMode==='focus'){
      sessionCount++;
      document.getElementById('sessionCount').textContent = sessionCount;
      socket.emit('send_message', {message: `⏱️ Completed a focus session! 🍅 Total: ${sessionCount} pomodoros`});
      // auto suggest break
      if(sessionCount%4===0){ setTimerMode('long'); }
      else { setTimerMode('short'); }
    } else {
      setTimerMode('focus');
    }
    return;
  }
  timerSeconds--;
  updateTimerDisplay();
}

function updateTimerDisplay(){
  const m = Math.floor(timerSeconds/60);
  const s = timerSeconds%60;
  const str = `${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
  document.getElementById('timerDisplay').textContent = str;
  document.getElementById('timerBadgeTime').textContent = str;

  // Update ring
  const circumference = 326.7;
  const progress = timerSeconds / timerMax;
  const offset = circumference * (1-progress);
  document.getElementById('timerRing').style.strokeDashoffset = offset;

  // Show badge in header when running
  document.getElementById('timerBadge').style.display = timerRunning ? 'flex' : 'none';
}

function resetTimer(){
  clearInterval(timerInterval);
  timerRunning = false;
  document.getElementById('timerStartBtn').textContent='▶ Start';
  timerSeconds = TIMER_MODES[currentMode].mins * 60;
  timerMax = timerSeconds;
  updateTimerDisplay();
}

// ─── Voice Input ────────────────────────────
let recognizing = false;
let recognition = null;

function toggleVoice(){
  if(!('webkitSpeechRecognition' in window || 'SpeechRecognition' in window)){
    alert('Voice input not supported in this browser. Try Chrome!');
    return;
  }
  if(recognizing){
    recognition.stop();
    return;
  }

  const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
  recognition = new SpeechRecognition();
  recognition.continuous = false;
  recognition.interimResults = false;
  recognition.lang = 'en-IN'; // Indian English + Tamil support

  recognition.onstart = () => {
    recognizing = true;
    document.getElementById('voiceBtn').classList.add('recording');
    document.getElementById('voiceBtn').textContent = '🔴';
  };

  recognition.onresult = (e) => {
    const transcript = e.results[0][0].transcript;
    msgInput.value = transcript;
    msgInput.dispatchEvent(new Event('input'));
  };

  recognition.onend = () => {
    recognizing = false;
    document.getElementById('voiceBtn').classList.remove('recording');
    document.getElementById('voiceBtn').textContent = '🎤';
  };

  recognition.onerror = (e) => {
    recognizing = false;
    document.getElementById('voiceBtn').classList.remove('recording');
    document.getElementById('voiceBtn').textContent = '🎤';
  };

  recognition.start();
}

  const avContent = isBot?'🤖':getInitials(data.username);
  const nameClass = isBot?'bot':'';
  const metaDir = isOwn?'style="justify-content:flex-end"':'';

  div.innerHTML=`
    <div class="msg-av ${avClass}">${avContent}</div>
    <div class="msg-c">
      <div class="msg-meta" ${metaDir}>
        <span class="msg-name ${nameClass}">${data.username}</span>
        <span class="msg-time">${data.timestamp}</span>
      </div>
      <div class="msg-bubble">${escapeHtml(data.message)}</div>
    </div>`;

  area.appendChild(div);
  area.scrollTop = area.scrollHeight;
}

function addSysMsg(text){
  const area = document.getElementById('messagesArea');
  const d = document.createElement('div');
  d.className='sys-msg';
  d.textContent=text;
  area.appendChild(d);
  area.scrollTop=area.scrollHeight;
}

function escapeHtml(t){
  return t.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/\n/g,'<br>');
}
